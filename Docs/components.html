<script>
(function() {
  // Active link
  const path = window.location.pathname.replace(/\/$/, '') || '/index.html';
  document.querySelectorAll('.nav-links a, .nav-mobile a').forEach(a => {
    const href = a.getAttribute('href').replace(/\/$/, '');
    if (path.endsWith(href) || (path === '' && href.includes('index'))) {
      a.classList.add('active');
    }
  });

  // Hamburger toggle
  const hamburger = document.getElementById('nav-hamburger');
  const mobileMenu = document.getElementById('nav-mobile');
  hamburger.addEventListener('click', function() {
    const isOpen = mobileMenu.classList.toggle('open');
    hamburger.setAttribute('aria-expanded', isOpen);
  });

  // Tournaments conditional — reads summary.json
  const DATA_BASE = 'https://raw.githubusercontent.com/jackstraw923/SureBetArcher/main/Docs/data/';
  fetch(DATA_BASE + 'summary.json')
    .then(r => r.json())
    .then(data => {
      if (data.tournaments_active) {
        const tournLink = '<a href="/tournaments.html">TOURNAMENTS</a>';
        // Insert after ODDS in desktop nav
        const oddsLink = document.querySelector('.nav-links a[href*="odds"]');
        if (oddsLink) oddsLink.insertAdjacentHTML('afterend', tournLink);
        // Insert after ODDS in mobile nav
        const oddsLinkM = document.querySelector('.nav-mobile a[href*="odds"]');
        if (oddsLinkM) oddsLinkM.insertAdjacentHTML('afterend', tournLink);
        // Re-run active state check
        const newLinks = document.querySelectorAll('.nav-links a[href*="tournaments"], .nav-mobile a[href*="tournaments"]');
        newLinks.forEach(a => {
          if (path.includes('tournaments')) a.classList.add('active');
        });
      }
    })
    .catch(() => {}); // Fail silently — tournaments hidden if fetch fails
})();
</script>

// Filter pill toggle — call with the container selector and a callback
function initFilterBar(barSelector, onFilter) {
  const bar = document.querySelector(barSelector);
  if (!bar) return;
  bar.querySelectorAll('.filter-pill').forEach(pill => {
    pill.addEventListener('click', function() {
      bar.querySelectorAll('.filter-pill').forEach(p => p.classList.remove('active'));
      this.classList.add('active');
      onFilter(this.textContent.trim());
    });
  });
}

function renderEquityCurve(canvasId, labels, values) {
  const ctx = document.getElementById(canvasId).getContext('2d');

      labels,
      datasets: [{
        data: values,
        borderColor: '#c9a84c',

        backgroundColor: 'rgba(45,212,191,0.08)',
        pointRadius: 0,
        pointHoverRadius: 4,
        pointHoverBackgroundColor: '#c9a84c',
        fill: true,

15 — Article Body (EdgeVault)
Long-form reading layout. DM Sans 300 at 15px, max-width 700px. H2 uses Bebas Neue. H3 uses DM Sans 500. Callout block is a gold left-border pull quote. Affiliate links use standard anchor style — gold color, underline. Never a button in article body.
article-body — rendered sample

What the Market Is Actually Saying

Closing line value is the strongest early signal a quantitative system can produce. It does not ask you to trust a backtest. It asks a simpler question: did the entry age well as the market absorbed more information?

How to think about CLV in plain language

Suppose ArcVest enters a position at a price that implies a lower probability than the closing market eventually implies. That means the entry was better than the final consensus price.

If a model cannot show that its entries beat the closing market over time, it is asking for trust without offering proof.

That is why Price Capture Alpha is surfaced in the ArcVest record and why the Performance Log exists.
